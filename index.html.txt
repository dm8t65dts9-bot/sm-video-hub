<script>
  const supabase = window.supabase.createClient(
    "https://urmhbibeksunrkmwfxyj.supabase.co",
    "sb_publishable_99fjx79j7zuGfdfy5wP0lA_ruQgzMUU"
  );

  let allVideos = [];

  function toEmbed(url) {
    const yt = url.match(/(?:youtube\.com\/watch\?v=|youtu\.be\/)([\w-]+)/);
    if (yt) return `https://www.youtube.com/embed/${yt[1]}`;
    const vm = url.match(/vimeo\.com\/(\d+)/);
    if (vm) return `https://player.vimeo.com/video/${vm[1]}`;
    return url;
  }

  function render(list) {
    document.getElementById("videos").innerHTML = list.length ? list.map(v => `
      <div class="card" onclick='openVideo(${JSON.stringify(v).replace(/'/g,"&#39;")})'>
        <img src="${v.thumbnail_url || "https://placehold.co/600x340/1e293b/38bdf8?text=SM+Video+Hub"}">
        <div class="card-body">
          <h3>${v.title}</h3>
          <p>${v.category || "General"}</p>
        </div>
      </div>
    `).join("") : "<p>কোনো ভিডিও পাওয়া যায়নি।</p>";
  }

  function openVideo(v) {
    document.getElementById("videoFrame").innerHTML =
      `<iframe src="${toEmbed(v.video_url)}" allowfullscreen></iframe>`;
    document.getElementById("videoTitle").textContent = v.title;
    document.getElementById("videoDesc").textContent = v.description || "";
    document.getElementById("player").classList.add("active");
  }

  function closePlayer() {
    document.getElementById("player").classList.remove("active");
    document.getElementById("videoFrame").innerHTML = "";
  }

  document.getElementById("search").addEventListener("input", e => {
    const q = e.target.value.toLowerCase();
    render(allVideos.filter(v =>
      v.title.toLowerCase().includes(q) ||
      (v.description || "").toLowerCase().includes(q)
    ));
  });

  async function loadVideos() {
    const { data, error } = await supabase
      .from("videos")
      .select("*")
      .eq("is_published", true)
      .order("created_at", { ascending: false });

    if (error) {
      document.getElementById("videos").innerHTML =
        `<p>ডেটা লোড করা যায়নি: ${error.message}</p>`;
      return;
    }
    allVideos = data || [];
    render(allVideos);
  }

  loadVideos();
</script>
