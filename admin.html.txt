<script>
  const supabase = window.supabase.createClient(
    "https://urmhbibeksunrkmwfxyj.supabase.co",
    "sb_publishable_99fjx79j7zuGfdfy5wP0lA_ruQgzMUU"
  );

  const ADMIN_EMAIL = "mahbubvai248@gmail.com";

  async function login() {
    const email = email.value;
    const password = password.value;
    const { error } = await supabase.auth.signInWithPassword({ email, password });
    document.getElementById("msg").textContent = error ? error.message : "Login successful";
    if (!error) showAdmin();
  }

  async function logout() {
    await supabase.auth.signOut();
    location.reload();
  }

  function showAdmin() {
    loginForm.style.display = "none";
    adminForm.style.display = "block";
  }

  async function addVideo() {
    const { error } = await supabase.from("videos").insert({
      title: title.value,
      category: category.value || "General",
      video_url: videoUrl.value,
      thumbnail_url: thumbnail.value,
      description: description.value,
      is_published: true
    });

    document.getElementById("msg").textContent = error
      ? "Error: " + error.message
      : "ভিডিও সফলভাবে যোগ হয়েছে!";

    if (!error) {
      title.value = category.value = videoUrl.value = thumbnail.value = description.value = "";
    }
  }

  supabase.auth.getUser().then(({ data }) => {
    if (data.user && data.user.email === ADMIN_EMAIL) showAdmin();
  });
</script>
